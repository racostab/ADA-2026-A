def detect_incendios(text):
    #Tokenizar el texto
    tokens = word_tokenize(text)

    #Determinar parte de la frase
    pos = pos_tag(tokens)

    #Ejecutar el código de entrenamiento
    ner = ne_chunk(pos)
    return ner
